local function getTotalWeight( ent )
    local weightTotal = ent:getPhysicsObject():getMass()
    for _, cEnt in ipairs( ent:getAllConstrained() ) do
        if cEnt:isValid() and cEnt:getPhysicsObject():isValid() then
            weightTotal = weightTotal + cEnt:getPhysicsObject():getMass()
        end
    end
    return weightTotal
end

local function wiremodApplyTorque( Ent, torque )
    if torque[1] == 0 and torque[2] == 0 and torque[3] == 0 then return end
    local tq = Vector( torque[1], torque[2], torque[3] )
    local torqueamount = tq:getLength()
    tq = Ent:localToWorld( tq ) - Ent:getPos()
    local off
    if math.abs( tq.x ) > torqueamount * 0.1 or math.abs( tq.z ) > torqueamount * 0.1 then
        off = Vector( -tq.z, 0, tq.x )
    else
        off = Vector( -tq.y, tq.x, 0 )
    end
    off = off:getNormalized() * torqueamount * 0.5
    local dir = ( tq:cross( off ) ):getNormalized()
    Ent:applyForceOffset( dir, off )
    Ent:applyForceOffset( dir * -1, off * -1 )
end

hook.add("Think", "ControlThink", function()
    if not Active or not isValid( baseEnt ) or not isValid( Driver ) then return end
    local time = timer.curtime()
    if time > lastLimitedTick + limitedTickDelay then
        wire.ports["ColorA"] = wire.ports["Color"] == colGreen and colRed or colGreen
        wire.ports["ColorB"] = wire.ports["Color"] == colRed and colGreen or colRed
        lastLimitedTick = time
    end
    if time > lastLimitedSmallTick + limitedSmallTickDelay then
        if MouseL == 1 and MouseR == 1 then
            local tarQ = CamAng:setR(0):getQuaternion()
            local curQ = baseEnt:getQuaternion()
            local Q = tarQ/curQ
            local vec = baseEnt:worldToLocal(Q:getRotationVector()+baseEnt:getPos())
            wiremodApplyTorque(baseEnt,(turnForce*vec - turnAccuracy*baseEnt:getAngleVelocity()+Vector(-(turnForce/5)*baseEnt:getAngleVelocity().z,0,0))*baseEnt:getInertia())
        end
        lastLimitedSmallTick = time
    end
    if time > lastLimitedLongTick + limitedLongTickDelay then
        wire.ports["Text"] = messages[math.random( #messages )]
        lastLimitedLongTick = time
    end
    local forceMult = ( Shift == 1 and ( defForceMult * 5 ) or defForceMult )
    if Shift == 1 and MouseL == 1 and MouseR == 1 and not ( Space == 1 or Alt == 1 ) then
        --basePhys:setVelocity( Vector( basePhys:getVelocity().x, basePhys:getVelocity().y, 0 ) )
    end
    if W == 1 then
        basePhys:applyForceCenter( baseEnt:getForward() * forceMult )
    end
    if A == 1 then
        if MouseL == 1 and MouseR == 1 then
            basePhys:applyForceCenter( baseEnt:getRight() * -forceMult ) 
        else 
            basePhys:addAngleVelocity( Vector( 0, 0, turnSpeed ) )
        end
    end
    if S == 1 then
        basePhys:applyForceCenter( baseEnt:getForward() * -forceMult  )
    end
    if D == 1 then
        if MouseL == 1 and MouseR == 1 then
            basePhys:applyForceCenter( baseEnt:getRight() * forceMult ) 
        else 
            basePhys:addAngleVelocity( Vector( 0, 0, -turnSpeed ) )
        end
    end
    if Space == 1 then
        if MouseL == 1 and MouseR == 1 then
            basePhys:applyForceCenter( game.getWorld():getUp() * forceMult ) 
        else 
            basePhys:applyForceCenter( baseEnt:getUp() * forceMult  )
        end
    end
    if Alt == 1 then
        basePhys:applyForceCenter( baseEnt:getUp() * -forceMult  )
    end
end )

                bootText = "Initializing Santa's Little Helper 2.3"

    local function isWithin(value, lower, upper)
        if upper < lower then

            upper = lower
            lower = lowerTemp
        end
        if value > lower and value < upper then
            return true
        else
            return false
        end
    end

hook.add("DrawHUD", "HudInfo", function()
    
    local x, y = render.getGameResolution()
    local midX, midY = x / 2, y / 2
    
    if not fullyBoot then
        render.setColor(colGreen)
        render.drawSimpleText(midX - 60, midY - 50, bootText .. bootAppend)
        return
    end
    
    -- Get pod data for HUD elements
    local podVel = vehicle:getVelocity()
    local podAng = vehicle:getAngles()
    local speed = math.round(podVel:getLength(), 1)
    local bearing = math.round(podAng.y, 0)
    
    -- Artificial Horizon
    local horizonSize = 400
    local horizonX = midX
    local horizonY = midY
    
    -- Get pitch and roll
    local pitch = -podAng.p
    local roll = podAng.r
    
    -- Clamp pitch for display
    local maxPitch = 60
    pitch = math.clamp(pitch, -maxPitch, maxPitch)
    
    -- Horizon line
    render.setColor(Color(0, 255, 0, 255))
    local horizonOffset = (pitch / maxPitch) * horizonSize / 2
    render.drawLine(horizonX - horizonSize/2, horizonY + horizonOffset, horizonX + horizonSize/2, horizonY + horizonOffset)
    
    -- Pitch lines
    for i = -6, 6 do
        if i ~= 0 then
            local pitchLineY = horizonY + horizonOffset - (i * 60)
            local lineLength = i % 2 == 0 and 40 or 20
            
            -- Only draw if within the horizon display area
            local horizonTop = horizonY - horizonSize/2
            local horizonBottom = horizonY + horizonSize/2
            
            if pitchLineY >= horizonTop and pitchLineY <= horizonBottom then
                render.drawLine(horizonX - lineLength/2, pitchLineY, horizonX + lineLength/2, pitchLineY)
                render.drawSimpleText(horizonX + horizonSize/2 + 5, pitchLineY - 8, tostring(i * 10))
            end
        end
    end
    
    -- Roll indicator (aircraft symbol)
    local rollRad = math.rad(roll)
    local wingLength = 90
    local noseLength = 90
    local centerX, centerY = horizonX, horizonY
    
    -- Draw aircraft symbol (rotated by roll)
    local noseX = centerX + math.sin(rollRad) * noseLength
    local noseY = centerY - math.cos(rollRad) * noseLength
    local leftWingX = centerX - math.cos(rollRad) * wingLength
    local leftWingY = centerY - math.sin(rollRad) * wingLength
    local rightWingX = centerX + math.cos(rollRad) * wingLength
    local rightWingY = centerY + math.sin(rollRad) * wingLength
    
    render.drawLine(centerX, centerY, noseX, noseY) -- Nose
    render.drawLine(centerX, centerY, leftWingX, leftWingY) -- Left wing
    render.drawLine(centerX, centerY, rightWingX, rightWingY) -- Right wing
    
    -- Roll marks
    for i = -3, 3 do
        if i ~= 0 then
            local markAngle = math.rad(i * 30)
            local markX1 = centerX + math.sin(rollRad + markAngle) * (horizonSize/2 - 10)
            local markY1 = centerY - math.cos(rollRad + markAngle) * (horizonSize/2 - 10)
            local markX2 = centerX + math.sin(rollRad + markAngle) * (horizonSize/2)
            local markY2 = centerY - math.cos(rollRad + markAngle) * (horizonSize/2)
            render.drawLine(markX1, markY1, markX2, markY2)
        end
    end
    
    local crosshairDistance = 500 -- Distance in front of vehicle (in units)
    local podPos = vehicle:getPos()
    local podForward = vehicle:getForward()
    
    -- Calculate position in front of vehicle
    local crosshairWorldPos = podPos + podForward * crosshairDistance
    
    -- Convert to screen coordinates
    local crosshairScreen = crosshairWorldPos:toScreen()
    
    -- Only draw if the point is on screen
    if crosshairScreen.visible then
        local crosshairX, crosshairY = crosshairScreen.x, crosshairScreen.y - 10
        local crosshairSize = 20
        local crosshairThickness = 2
        
        render.setColor(Color(0, 255, 0, 255))
        
        -- Horizontal lines
        render.drawRect(crosshairX - crosshairSize, crosshairY - crosshairThickness/2, crosshairSize * 2, crosshairThickness)
        render.drawRect(crosshairX - crosshairSize, crosshairY - crosshairSize * 2, crosshairSize * 2, crosshairThickness)
        -- Vertical line
        render.drawRect(crosshairX - crosshairThickness/2, crosshairY - crosshairSize * 2, crosshairThickness, crosshairSize * 2)
    end
    
    -- Speed display (top left)
    render.drawSimpleText(midX - 300, midY, "SPD: " .. tostring(speed) .. " u/s")
    
    local compassY = 20 -- Position from top
    local compassCenterX = midX
    
    -- Box dimensions
    local boxWidth = 60
    local boxHeight = 30
    local boxX = compassCenterX - boxWidth/2
    local boxY = compassY
    
    -- Current bearing
    local currentBearing = bearing
    
    -- Calculate pixel width per degree for the ribbon
    local pixelsPerDegree = 4
    
    -- Draw the ribbon background (extending beyond the box)
    local ribbonWidth = boxWidth -- Ribbon extends beyond box
    local ribbonHeight = boxHeight
    render.setColor(Color(0, 0, 0, 200))
    render.drawRect(compassCenterX - ribbonWidth/2, boxY, ribbonWidth, ribbonHeight)
    
    -- Draw numbers on the ribbon
    local startBearing = currentBearing - (ribbonWidth / 2 / pixelsPerDegree)
    local endBearing = currentBearing + (ribbonWidth / 2 / pixelsPerDegree)
    
    for bearingTick = math.floor(startBearing), math.ceil(endBearing) do
        -- Normalize bearing to 0-360
        local normalizedBearing = bearingTick % 360
        if normalizedBearing < 0 then normalizedBearing = normalizedBearing + 360 end
        
        -- Calculate X position relative to center
        local bearingOffset = bearingTick - currentBearing
        local numberX = compassCenterX + (bearingOffset * pixelsPerDegree)
        
        -- Only draw if within ribbon bounds
        if numberX >= compassCenterX - ribbonWidth/2 and numberX <= compassCenterX + ribbonWidth/2 then
            -- Draw number every 10 degrees
            if normalizedBearing % 10 == 0 then
                local bearingText = string.format("%03d", normalizedBearing)
                local textWidth = render.getTextSize(bearingText)
                
                render.setColor(Color(0, 255, 0, 200)) -- Green for ribbon numbers
                render.drawSimpleText(numberX - textWidth/2, boxY + boxHeight/2 - 8, bearingText)
            end
            
            -- Draw tick marks every 5 degrees
            if normalizedBearing % 5 == 0 then
                local tickLength = (normalizedBearing % 10 == 0) and 8 or 4
                render.setColor(Color(0, 255, 0, 150))
                render.drawLine(numberX, boxY, numberX, boxY + tickLength)
                render.drawLine(numberX, boxY + boxHeight, numberX, boxY + boxHeight - tickLength)
            end
        end
    end
    
    -- Draw the box border (window that shows current heading)
    render.setColor(Color(0, 255, 0, 255)) -- Green border
    render.drawLine(boxX, boxY, boxX + boxWidth, boxY) -- Top
    render.drawLine(boxX, boxY + boxHeight, boxX + boxWidth, boxY + boxHeight) -- Bottom
    render.drawLine(boxX, boxY, boxX, boxY + boxHeight) -- Left
    render.drawLine(boxX + boxWidth, boxY, boxX + boxWidth, boxY + boxHeight) -- Right
    
    -- Original player detection HUD
    if player():isInPVE() then
        for i, ply in pairs(playersTemp) do
            if not isValid(ply) or ply == player() then continue end
            render.setColor(ply:isInPVE() and colGreen or colRed)
            local pos = ply:getPos():toScreen()
            if isWithin(pos.x, midX - withinPx.x, midX + withinPx.x) and isWithin(pos.y, midY - withinPx.y, midY + withinPx.y) then
                text = ply:getName()
            else
                text = ply:isInPVE() and "NICE PLAYER" or "NAUGHTY PLAYER"
            end
            render.drawSimpleText(pos.x, pos.y, text)
        end
    end
end) 
    
end )
