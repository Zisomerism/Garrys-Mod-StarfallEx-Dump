--@name vehicles_util/glide projectile mortar
--@author desu
--@shared

-- This chip can control a glide projectile turret from a pod controller in several modes:

-- Default Mode:
-- The turret will match the pitch and yaw of the pod controller.

-- Hit Position Aiming Mode:
-- The turret will aim at the aim hit position of the pod controller.

-- Auto Aim Mode:
-- The turret will automatically aim at the closest target in a sphere around current aim point.
-- If the turret cannot find a direct line of sight to a target, it will attempt to aim with a ballistic trajectory to the target.
-- If the turret is able to find a target, it will fire a projectile at the target.
-- The turret will switch targets if it cannot get a good shot at the current target.

-- Usage:
-- Spawn this chip and a Pod Controller on a chair
-- Wire "PodController [ENTITY]" from this chip to "Create Entity [ENTITY]" on a Pod Controller
-- Spawn a Glide Projectile Launcher and some kind of prop to attach it to
-- Wire "Turret [ENTITY]" from this chip to "Create Entity [ENTITY]" on the Glide Projectile Launcher
-- Wire "TurretBody [ENTITY]" from this chip to "Create Entity [ENTITY]" on the prop the Glide Projectile Launcher will be attached to
-- Using a parent tool of some kind, parent the Glide Projectile Launcher to the prop you wired TurretBody to.
-- Make sure the Pod Controller is connected to the chair and get in...

-- edit these vars if you want ------------------------------------------------
local defaultSpeed = 4000

-- RMB artillery cam settings
local artyCamPitch = 80
local artyCamOffsetY = 400
local artyCamOffsetZ = 2000

-- auto aim settings
local searchRadius = 8192
local inRangeDistSqr = 1024
local autoAimThinkDelay = 1
local autoAimArcShotDelay = 6
local autoAimTargetSwitchDelay = 20
local autoAimSpeedFloor = 600

-- probably don't edit these vars ---------------------------------------------
-- used for trajectory prediction
local predictionSteps = 128 
local predictionDt = 0.08

-- glide projectile turret settings
local glideTurretDelay = 3
local glideTurretDamage = 200
local glideTurretRadius = 500
local glideTurretSpeed = defaultSpeed
local glideTurretGravity = 600

-- auto aim internal vars
local searchResults = {}
local targetInRange = false
local autoAimThinkTime = 0
local autoAimArcShotTime = 0 
local autoAimTargetSwitchTime = 0 
local autoAimArcPreferred = 0

local colRed, colGreen, colBlue, colYellow = Color( 255, 50, 50 ), Color( 50, 255, 50 ), Color( 50, 200, 255 ), Color( 255, 255, 50 )

local turret, turretBody, podController, isActive, Driver
local W, A, S, D, Space, Alt, Shift, MouseL, MouseR, toggleHitPosAiming, toggleDefendArea, AimPos
local projectileTemp
local predictedHitPos = Vector( 0, 0, 0 )
local projParamTime = 0
local projParamDelay = 0.08
if SERVER then

    local function outOfBudget( ratio )
        ratio = ratio or 0.9
        return cpuAverage() > cpuMax() * ratio or cpuUsed() > cpuMax() * ratio
    end

    local function setTurretCol()
        if toggleDefendArea == 1 then
            turret:setColor( colBlue )
        elseif toggleHitPosAiming == 1 and isActive == 1 then
            turret:setColor( colYellow )
        elseif isActive == 1 then
            turret:setColor( colGreen )
        else
            turret:setColor( colRed )
        end
    end

    local function angleDeltaDeg( a, b )
        local d = ( a - b ) % 360
        if d > 180 then d = d - 360 end
        return d
    end

    local function buildProjectileTraceFilter()
        local f = { chip() }
        if isValid( turret ) then f[#f + 1] = turret end
        if isValid( turretBody ) then f[#f + 1] = turretBody end
        if isValid( Driver ) then f[#f + 1] = Driver end
        if isValid( projectileTemp ) then f[#f + 1] = projectileTemp end
        return f
    end

    local predSim = {
        active = false,
        i = 1,
        pos = Vector( 0, 0, 0 ),
        vel = Vector( 0, 0, 0 ),
        g = Vector( 0, 0, 0 ),
        lastPos = Vector( 0, 0, 0 ),
        filter = nil
    }

    local function updatePredictedHitPos()
        if not isValid( turret ) then
            predSim.active = false
            return
        end

        if not predSim.active then
            local dir = turret:getAngles():getUp()
            predSim.pos = turret:getPos() + dir * 10
            predSim.vel = dir * glideTurretSpeed
            predSim.g = Vector( 0, 0, -glideTurretGravity )
            predSim.lastPos = predSim.pos
            predSim.filter = buildProjectileTraceFilter()
            predSim.i = 1
            predSim.active = true
        end

        local pos, vel, g = predSim.pos, predSim.vel, predSim.g
        local halfG = g * 0.5
        local lastPos = predSim.lastPos
        local filter = predSim.filter or buildProjectileTraceFilter()

        for i = predSim.i, predictionSteps do
            local t = i * predictionDt
            if outOfBudget( 0.5 ) then
                predSim.i = i + 1
                predSim.lastPos = lastPos
                return
            end

            local nextPos = pos + vel * t + halfG * ( t * t )
            local tr = trace.line( lastPos, nextPos, filter )
            if tr.Hit then
                predictedHitPos = tr.HitPos
                predSim.active = false
                return
            end
            lastPos = nextPos

        end

        predictedHitPos = lastPos
        predSim.active = false
    end

        wire.ports["Damage"] = glideTurretDamage
        wire.ports["Radius"] = glideTurretRadius

        wire.ports["Gravity"] = glideTurretGravity
    end

    local camHolo, camActive
    local turretOffset = { p = nil, y = nil }
    local function initCam()
        camHolo = hologram.create( chip():getPos(), Angle( 0, 0, 0 ), "models/hunter/plates/plate.mdl", Vector( 0, 0, 0 ) )
        if isValid( turret ) and isValid( turretBody ) then
            local muzzleAng = turret:getAngles():getUp():getAngle()
            local bodyAng = turretBody:getAngles()
            turretOffset.p = angleDeltaDeg( muzzleAng.p, bodyAng.p )
            turretOffset.y = angleDeltaDeg( muzzleAng.y, bodyAng.y )
        end
    end

    local function viewCam( value )
        if value == 1 then 
            camActive = 1
            Driver:setViewEntity( camHolo )
        else
            camActive = 0
            Driver:setViewEntity( Driver )
        end
    end

    local gunCam
    local function fire( value )
        if value == nil then
            fire( 1 )
            timer.simple( 0.05, function() fire( 0 ) end )
            return
        end
        MouseL = value
        wire.ports["Fire"] = MouseL
        if value ~= 1 or not isValid( Driver ) or not isValid( turret ) then return end
        timer.simple( 0.1, function()
            for _, ent in pairs( find.inSphere( turret:getPos(), 512 ) ) do
                if isValid( ent ) and ent:getClass() == "glide_projectile" then
                    projectileTemp = ent
                    if gunCam == 1 then
                        Driver:setViewEntity( ent )
                    end
                end
            end
        end )
    end

    local function clamp( x, lo, hi )
        return x < lo and lo or ( x > hi and hi or x )
    end

    -- d = distance, h = height difference, v0 = velocity
    local function ballisticHighArcPitchDeg( d, h, v0 )
        if outOfBudget( 0.7 ) then return nil end
        if d < 1 or v0 < 1 then return nil end
        local v2 = v0 * v0
        local inner = v2 * v2 - glideTurretGravity * ( glideTurretGravity * d * d + 2 * h * v2 )
        -- Near the minimum muzzle speed for (d,h), float error makes inner slightly negative. Treat as 0.
        if inner < 0 and inner > -0.01 then inner = 0 end
        if inner < 0 then return nil end
        local tanTheta = ( v2 + math.sqrt( inner ) ) / ( glideTurretGravity * d )
        return -math.deg( math.atan( tanTheta ) )
    end

    local pid = {
        kp_p = 6.0, ki_p = 0.0, kd_p = 1.5,
        kp_y = 6.0, ki_y = 0.0, kd_y = 1.5,
        iClamp = 35,
        outClamp = 25,
        lastT = nil,
        iP = 0, iY = 0,
        lastAngPredY = nil,
        lastMeasP = nil,
        wasArc = false,
    }

    local function PIDControl( targetPos )
        if toggleHitPosAiming ~= 1 then return end
        if not isValid( turret ) or not isValid( turretBody ) then return end
        if not targetPos then return turretBody:getAngles() end
        if outOfBudget( 0.6 ) then return turretBody:getAngles() end

        local dist = targetPos:getDistanceSqr( predictedHitPos )
        if dist < inRangeDistSqr then
            targetInRange = true
            return turretBody:getAngles()
        else
            targetInRange = false
        end

        local cur = turretBody:getAngles()
        local origin = turret:getPos()
        local toTgt = targetPos - origin
        if toTgt:getLengthSqr() < 1 then return cur end

        local angTgt = toTgt:getNormalized():getAngle()
        local d = Vector( toTgt.x, toTgt.y, 0 ):getLength()
        local h = toTgt.z

        -- Arc mode: direct analytic set using calibrated turretBody→muzzle offset.
        -- Physics is exact (no drag, known gravity) so the formula and the sim agree perfectly.
        if autoAimArcPreferred == 1 then
            pid.wasArc = true
            if not turretOffset.p then
                return cur
            end
            local pArc = ballisticHighArcPitchDeg( d, h, glideTurretSpeed )
            if not pArc then
                return cur
            end
            return Angle( pArc - turretOffset.p, angTgt.y - turretOffset.y, cur.r )
        end

        -- Non-arc: PID drives predicted impact direction toward target LOS.
        -- Reset state cleanly when transitioning out of arc.
        if pid.wasArc then
            pid.wasArc = false
            pid.iP, pid.iY = 0, 0
            pid.lastT = nil
            pid.lastMeasP = nil
            pid.lastAngPredY = nil
        end

        local t = timer.curtime()
        local dt = pid.lastT and ( t - pid.lastT ) or 0
        pid.lastT = t
        if dt <= 0 or dt > 0.25 then dt = 0.02 end

        local toPred = predictedHitPos - origin
        if toPred:getLengthSqr() < 1 then return cur end

        local angPred = toPred:getNormalized():getAngle()

        local errP = angleDeltaDeg( angTgt.p, angPred.p )
        local errY = angleDeltaDeg( angTgt.y, angPred.y )

        -- Lateral yaw feedforward: converts XY miss distance into an extra yaw correction.
        -- Helps at long range where a small angular error still means a large lateral miss.
        if d > 1 then
            local miss = targetPos - predictedHitPos
            -- 2D cross product of aim direction (toTgt.xy / d) with miss.xy gives signed lateral distance.
            local lateral = ( toTgt.x * miss.y - toTgt.y * miss.x ) / d
            errY = errY + clamp( math.deg( math.atan2( lateral, d ) ), -15, 15 )
        end

        pid.iP = clamp( pid.iP + errP * dt, -pid.iClamp, pid.iClamp )
        pid.iY = clamp( pid.iY + errY * dt, -pid.iClamp, pid.iClamp )

        local dErrP, dErrY = 0, 0
        if pid.lastMeasP ~= nil then
            dErrP = -angleDeltaDeg( angPred.p, pid.lastMeasP ) / dt
        end
        pid.lastMeasP = angPred.p
        if pid.lastAngPredY ~= nil then
            dErrY = -angleDeltaDeg( angPred.y, pid.lastAngPredY ) / dt
        end
        pid.lastAngPredY = angPred.y

        local cmdP = pid.kp_p * errP + pid.ki_p * pid.iP + pid.kd_p * dErrP
        local cmdY = pid.kp_y * errY + pid.ki_y * pid.iY + pid.kd_y * dErrY

        local errMag = math.max( math.abs( errP ), math.abs( errY ) )
        local speedScale = clamp( errMag / 20, 0.25, 3.0 )
        local maxRate = pid.outClamp * speedScale
        cmdP = clamp( cmdP, -maxRate, maxRate )
        cmdY = clamp( cmdY, -maxRate, maxRate )

        return Angle( cur.p + cmdP * dt, cur.y + cmdY * dt, cur.r )
    end

    local function cleanupDriver( driver )
        net.start( "CleanupDriverClient" )
        net.send( Driver, false )
        enableHud( Driver, false )
        setTurretCol()
        driver:clearPVSPoints()
    end

    local function newTargets()
        local results = {}
        for _, v in pairs( find.inSphere( predictedHitPos, searchRadius ) ) do
            if not v:isNPC() or v:getHealth() <= 0 then continue end
            if v:isValid() then
                results[#results + 1] = v
            end
        end
        searchResults = find.sortByClosest( results, predictedHitPos )
    end 

    local function toggleHitPosAim( value )
        toggleHitPosAiming = value or ( toggleHitPosAiming == 1 and 0 ) or 1
        wire.ports["ToggleHitPosAiming"] = toggleHitPosAiming
        if value == 1 and toggleDefendArea == 1 then
            toggleDefendArea = 0
            wire.ports["ToggleDefendArea"] = toggleDefendArea
        end
        setTurretCol()
    end

    local function toggleDefend( value )
        local temp = value or ( toggleDefendArea == 1 and 0 ) or 1
        toggleHitPosAim( temp )
        if toggleHitPosAiming == 1 then
            newTargets()
        end
        toggleDefendArea = temp
        wire.ports["ToggleDefendArea"] = toggleDefendArea
        setTurretCol()
    end

    local function setSpeed( value )
        glideTurretSpeed = value or glideTurretSpeed
        wire.ports["Speed"] = glideTurretSpeed
    end

    local function adjustSpeed( positive )
        local v = Shift == 1 and 2000 or 100
        setSpeed( clamp( glideTurretSpeed + ( positive and v or -v ), 0, 20000 ) )
    end

    wire.adjustInputs( {"Turret", "TurretBody", "PodController", "Active", "Driver", "W", "A", "S", "D", "R", "Space", "Alt", "Shift", "PrevWeapon", "NextWeapon", "Mouse1", "Mouse2", "Light", "AimPos"}, {"entity", "entity", "entity", "number", "entity", "number", "number", "number", "number", "number", "number", "number", "number", "number", "number", "number", "number", "number", "vector"} )
    hook.add("Input", "HandleEntityInput", function(name, value)
        if name == "Turret" and isValid( value ) then
            turret = value
            turret:setColor( colRed )
            initTurret()
        elseif name == "TurretBody" and isValid( value ) then
            turretBody = value
        elseif name == "PodController" and isValid( value ) then
            podController = value
            initPod()
        elseif name == "Active" then
            isActive = value
        elseif name == "Driver" then
            if isValid( value ) then
                Driver = value
                initCam()
                net.start( "InitDriverClient" )
                net.writeEntity( turret )
                net.writeEntity( isValid( turretBody ) and turretBody or turret )
                net.send( Driver, false )
                enableHud( Driver, true )
                setTurretCol()
                projParamTime = timer.curtime()
            elseif isValid( Driver ) then
                cleanupDriver( Driver )
            end
        elseif name == "W" then
            W = value
        elseif name == "A" then
            A = value
        elseif name == "S" then
            S = value
        elseif name == "D" then
            D = value
        elseif name == "R" and value == 1 then
            toggleHitPosAim()
        elseif name == "Space" then
            Space = value
        elseif name == "Alt" then
            Alt = value
        elseif name == "Shift" then
            Shift = value
            gunCam = Shift
        elseif name == "Mouse1" then
            fire( value )
        elseif name == "Mouse2" and isValid( Driver ) then
            viewCam( value )
        elseif name == "PrevWeapon" then
            if value == 1 then adjustSpeed( true ) end
        elseif name == "NextWeapon" then
            if value == 1 then adjustSpeed( false ) end
        elseif name == "Light" and value == 1 then
            toggleDefend()
        elseif name == "AimPos" and value ~= nil then
            AimPos = value
        end
    end)

    wire.adjustOutputs( {"Fire", "Delay", "Damage", "Radius", "Speed", "Gravity", "ToggleHitPosAiming", "ToggleDefendArea"}, {"number", "number", "number", "number", "number", "number", "number", "number"} )
    
    hook.add("Think", "ControlThink", function()
        if isActive ~= 1 and toggleHitPosAiming ~= 1 then return end
        
        if isValid( turretBody ) then
            if outOfBudget() then return end
            if toggleDefendArea == 1 then
                local target = searchResults[1]
                if isValid( target ) then
                    local targetPos = target:getPos()
                    -- Enable arc + shell speed before PID so the same tick aims with correct arc/glideTurretSpeed.
                    if timer.curtime() >= autoAimArcShotTime and autoAimArcPreferred ~= 1 then
                        autoAimArcPreferred = 1
                        local origin = turret:getPos()
                        local r = targetPos - origin
                        local d = Vector( r.x, r.y, 0 ):getLength()
                        local h = r.z

                        local speed = defaultSpeed
                        local v2 = glideTurretGravity * ( h + math.sqrt( d * d + h * h ) )
                        if v2 > 0 then
                            speed = math.sqrt( v2 )
                        end

                        speed = clamp( speed, autoAimSpeedFloor, defaultSpeed )
                        setSpeed( speed )
                    end
                    local outang = PIDControl( targetPos )
                    turretBody:setAngles( outang )
                    if timer.curtime() >= autoAimTargetSwitchTime then
                        autoAimTargetSwitchTime = timer.curtime() + autoAimTargetSwitchDelay
                        autoAimArcShotTime = timer.curtime() + autoAimArcShotDelay
                        autoAimArcPreferred = 0
                        setSpeed( defaultSpeed )
                        newTargets()
                    end
                    if timer.curtime() >= autoAimThinkTime and targetInRange then
                        autoAimThinkTime = timer.curtime() + autoAimThinkDelay
                        autoAimTargetSwitchTime = timer.curtime() + autoAimTargetSwitchDelay
                        fire()
                    end
                else
                    if timer.curtime() >= autoAimThinkTime then
                        autoAimThinkTime = timer.curtime() + autoAimThinkDelay
                        autoAimArcShotTime = timer.curtime() + autoAimArcShotDelay
                        autoAimArcPreferred = 0
                        setSpeed( defaultSpeed )
                        newTargets()
                    end
                end
            elseif toggleHitPosAiming == 1 then
                if AimPos ~= nil then
                    local outang = PIDControl( AimPos )
                    turretBody:setAngles( outang )
                end
            else
                if not isValid( Driver ) then return end
                local eyes = Driver:getEyeAngles()
                -- Subtract the body→muzzle offset so the barrel points where the driver is looking.
                turretBody:setAngles( turretOffset.p and Angle( eyes.p - turretOffset.p, eyes.y - turretOffset.y, eyes.r ) or eyes )
            end
            if not isValid( Driver ) then return end
            camHolo:setPos( ( predictedHitPos - Angle( 0, Driver:getEyeAngles().y, 0 ):getForward() * artyCamOffsetY ) + Vector( 0, 0, artyCamOffsetZ ) ) 
            camHolo:setAngles( Angle( artyCamPitch, Driver:getEyeAngles().y, 0 ) )
            if camActive and Driver:isHUDActive() then
                Driver:setPVSPoint( 1, predictedHitPos )
            end
        end

        if not isValid( turret ) then return end
        local t = timer.curtime()
        if t >= projParamTime + projParamDelay then
            projParamTime = t
            local simOk = pcall( updatePredictedHitPos )
            if not simOk then
                predSim.active = false
                print( "[glide mortar] CPU spike during trajectory sim — skipped this tick" )
            else
                net.start( "ProjectileLauncherParams" )
                net.writeFloat( glideTurretSpeed )
                net.writeAngle( turret:getAngles() )
                net.send( Driver, false )
            end
        end
    end )

end
    
if CLIENT then
    local launcherSpeed, launcherAngles, launcherPos = 10000, Angle( 0, 0, 0 ), Vector( 0, 0, 0 )
    local trajTurret, trajTurretBody
    local tracerCol = Color(255, 255, 255)
    local font = render.createFont('FontAwesome', 18, 800, false, false, true, true, 0, false, 0)

    local function buildClientTraceFilter()
        local f = { chip(), player() }
        if isValid( trajTurret ) then f[#f + 1] = trajTurret end
        if isValid( trajTurretBody ) and trajTurretBody ~= trajTurret then f[#f + 1] = trajTurretBody end
        return f
    end

    net.receive( "ProjectileLauncherParams", function()
        launcherSpeed = net.readFloat()
        launcherAngles = net.readAngle()
        if isValid( trajTurret ) then
            launcherPos = trajTurret:getPos()
        end
    end )

    net.receive( "InitDriverClient", function()
        trajTurret = net.readEntity()
        trajTurretBody = net.readEntity()

        hook.add( "PostDrawOpaqueRenderables", "GlideProjTrajHud", function()
            local dir = launcherAngles:getUp()
            local pos = launcherPos + dir * 10
            local vel = dir * launcherSpeed
            local g = Vector( 0, 0, -glideTurretGravity )
            local lastPos = pos
            local filter = buildClientTraceFilter()
            render.setColor( colGreen )

            for i = 1, predictionSteps do
                local t = i * predictionDt
                local nextPos = pos + vel * t + g * ( 0.5 * t * t )
                local tr = trace.line( lastPos, nextPos, filter )
                local drawTo = tr.Hit and tr.HitPos or nextPos
                render.draw3DLine( lastPos, drawTo, tracerCol )
                if tr.Hit then break end
                lastPos = nextPos
            end
        end )

        hook.add( "DrawHUD", "HudInfo", function()
            render.setFont(font)
            local w, h = render.getGameResolution()
            local margin = 16
            local lineH = 22
            local titleGap = 6
            local rows = {
                { text = "Controls", color = colGreen, extraBelow = titleGap },
                { text = "Left Mouse = Fire", color = colGreen },
                { text = "Shift + Left Mouse = Shell Camera", color = colGreen },
                { text = "Right Mouse = Artillery Camera", color = colGreen },
                { text = "MWheel Up/Down = Adjust shell velocity (+/- 100)", color = colGreen },
                { text = "Shift + MWheel Up/Down = Adj shell vel (+/- 2000)", color = colGreen },
                { text = "R (Toggle) = Hit Position Aiming", color = colGreen },
                { text = "F (Toggle) = Auto Aiming", color = colGreen },
            }
            local totalH = 0
            for _, row in ipairs( rows ) do
                totalH = totalH + lineH + ( row.extraBelow or 0 )
            end
            local lineY = h - margin - totalH
            for _, row in ipairs( rows ) do
                render.setColor( row.color )
                local tw = render.getTextSize( row.text )
                render.drawSimpleText( w - margin - tw, lineY, row.text )
                lineY = lineY + lineH + ( row.extraBelow or 0 )
            end

            local speedRight = w * 0.7
            local elevGap = 4
            local blockH = lineH * 2 + elevGap
            local blockTopY = h * 0.5 - blockH * 0.5

            local elevStr
            if isValid( trajTurret ) then
                elevStr = string.format( "Elevation: %.1f deg", trajTurret:getAngles():getUp():getAngle().p )
            else
                elevStr = "Elevation: __"
            end
            local speedStr = "Shell Velocity: " .. tostring( math.floor( launcherSpeed + 0.5 ) ) .. " u/s"
            render.setColor( colGreen )
            local elevTw = render.getTextSize( elevStr )
            local speedTw = render.getTextSize( speedStr )
            render.drawSimpleText( speedRight - elevTw, blockTopY, elevStr )
            render.drawSimpleText( speedRight - speedTw, blockTopY + lineH + elevGap, speedStr )
        end )

        enableHud( player(), true )
    end )

    net.receive( "CleanupDriverClient", function()
        hook.remove( "PostDrawOpaqueRenderables", "GlideProjTrajHud" )
        hook.remove( "DrawHUD", "HudInfo" )
        enableHud( player(), false )
        trajTurret, trajTurretBody = nil, nil
    end )
